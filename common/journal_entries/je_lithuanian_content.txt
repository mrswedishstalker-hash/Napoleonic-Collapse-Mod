je_suvalkija_suwalki = {
    group = je_group_historical_content
    icon = "gfx/interface/icons/event_icons/event_map.dds"
    should_be_pinned_by_default_uninvolved_or_context = yes

    is_shown_when_inactive = {
        is_player = yes
        NOT = { has_variable = je_suvalkija_suwalki_ended }
		OR = {
            c:LIT ?= THIS
            c:POL ?= THIS
        }
    }

    possible = {
		OR = {
            c:LIT ?= THIS
            c:POL ?= THIS
        }
        NOT = { has_variable = je_suvalkija_suwalki_ended }
    }

    scripted_button = je_suvalkija_suwalki_serfdom
    scripted_button = je_suvalkija_suwalki_unify

    complete = {
        owns_entire_state_region = STATE_KAUNAS
    }

    on_complete = {
        trigger_event = { id = suvalkija_suwalki.1 }
        add_loyalists = {
            value = 0.2
            strata = middle
        }
        set_variable = lit_enacted_serfdom_ended
    }
}
