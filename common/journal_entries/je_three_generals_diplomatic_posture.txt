je_three_generals_diplomatic_posture = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"
	group = je_group_historical_content
 
	is_shown_when_inactive = {
		exists = c:FRA
		c:FRA ?= THIS
		NOT = { has_variable = three_generals_resolved }
		has_variable = three_generals_startup
	}
 
	possible = {
		exists = c:FRA
		c:FRA ?= THIS
		NOT = { has_variable = three_generals_resolved }
		has_variable = three_generals_startup
	}
 
	timeout = 3650
 
	complete = {
		OR = {
			# PATH 1: Jean
			AND = {
				custom_tooltip = { text = posture_jean_army_tt posture_jean_army = yes }
				custom_tooltip = { text = posture_jean_navy_tt posture_jean_navy = yes }
				OR = {
					custom_tooltip = { text = posture_jean_gbr_tt posture_jean_defeat_gbr = yes }
					custom_tooltip = { text = posture_jean_power_tt posture_jean_top_power = yes }
				}
			}
			# PATH 2: Bertrand
			AND = {
				custom_tooltip = { text = posture_bertrand_bloc_tt posture_bertrand_bloc = yes }
				custom_tooltip = { text = posture_bertrand_rivals_tt posture_bertrand_rivals = yes }
				OR = {
					custom_tooltip = { text = posture_bertrand_merc_tt posture_bertrand_mercantilism = yes }
					custom_tooltip = { text = posture_bertrand_grain_tt posture_bertrand_grain = yes }
				}
			}
			# PATH 3: Reille
			AND = {
				custom_tooltip = { text = posture_reille_gdp_tt posture_reille_gdp = yes }
				custom_tooltip = { text = posture_reille_invest_tt posture_reille_investment = yes }
				OR = {
					custom_tooltip = { text = posture_reille_trade_tt posture_reille_trade_privileges = yes }
					custom_tooltip = { text = posture_reille_wide_tt posture_reille_wide_investment = yes }
				}
			}
		}
	}

	on_complete = {
	if = {
		limit = { posture_jean_path_met = yes }
		change_variable = { name = jean_points add = 3 }
		# any other Jean-only reward goes here, e.g. add_modifier
	}
	else_if = {
		limit = { posture_bertrand_path_met = yes }
		change_variable = { name = bertrand_points add = 33 }
	}
	else_if = {
		limit = { posture_reille_path_met = yes }
		change_variable = { name = reille_points add = 33 }
	}
	three_generals_sync_bars_effect = yes
}
}
