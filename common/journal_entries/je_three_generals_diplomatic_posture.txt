je_three_generals_diplomatic_posture = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"
	group = je_group_historical_content
 
	is_shown_when_inactive = {
		exists = c:FRA
		c:FRA ?= THIS
		NOT = { has_variable = three_generals_resolved }
		has_variable = three_generals_startup
	}
 
	possible = {
		exists = c:FRA
		c:FRA ?= THIS
		NOT = { has_variable = three_generals_resolved }
		has_variable = three_generals_startup
	}
 
	timeout = 3650
 
	complete = {
		OR = {
 
			# ------------------------------------------------------------
			# Path 1: Jean's Military Economy
			# ------------------------------------------------------------
			AND = {
                custom_tooltip = {
				text = je_jeans_economy_requirements_tt
 
				NOT = { any_country = { army_size > root.army_size } }
				NOT = { any_country = { navy_size > root.navy_size } }
				OR = {
					has_variable = generals_defeated_gbr
					AND = {
						country_rank >= rank_value:great_power
						global_country_ranking = 1
					}
				}
			}
            }
			# ------------------------------------------------------------
			# Path 2: Bertrand's Agrarian Economy
			# ------------------------------------------------------------
			AND = {
            custom_tooltip = {
				text = je_bertrands_economy_requirements_tt
 
				power_bloc ?= { has_identity = identity:identity_sovereign_empire }
 
				# Rivalries (initiated by us) with GBR, AUT and TUR
				exists = c:GBR
				exists = c:AUT
				exists = c:TUR
				has_diplomatic_pact = { who = c:GBR type = rivalry is_initiator = yes }
				has_diplomatic_pact = { who = c:AUT type = rivalry is_initiator = yes }
				has_diplomatic_pact = { who = c:TUR type = rivalry is_initiator = yes }
 
				OR = {
					has_law = law_type:law_mercantilism
					# Grain stand-in for "largest grain producer".
					# vanilla has no "highest producer" trigger; tune this number.
					AND =
                    market_capital.market = {
						mg:grain = { market_goods_production >= 500 }
					}
				}
			}
            }
			# ------------------------------------------------------------
			# Path 3: Reille's Liberal Economics
			# ------------------------------------------------------------
			AND = {
            custom_tooltip = {
				text = je_reille_economy_requirements_tt
 
				gdp_ranking <= 3
 
				# We hold foreign investment rights in at least one country
				any_country = {
					NOT = { this = root }
					root = {
						has_treaty_foreign_investment_rights_with = { TARGET = prev }
					}
				}
 
				# ...and also hold trade privileges, or investment rights in 2+ countries
				OR = {
					any_country = {
						NOT = { this = root }
						root = {
							any_scope_treaty = {
								binds = prev.prev
								any_scope_article = {
									has_type = trade_privilege
									source_country = prev.prev.prev
								}
							}
						}
					}
					any_country = {
						limit = {
							NOT = { this = root }
							root = {
								has_treaty_foreign_investment_rights_with = { TARGET = prev }
							}
						}
						count >= 2
					}
				}
			}
		}
	}
}
	on_complete = {
		trigger_event = je_generals.2
	}
}
