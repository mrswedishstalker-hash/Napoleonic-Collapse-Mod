law_guided_technocracy = {
	group = lawgroup_distribution_of_power
	parent = law_technocracy        # ideological stances + counts for has_law_or_variant
	icon = "gfx/interface/icons/law_icons/technocracy.dds"
	progressiveness = -50

	# Only available from a monarchy (same pattern as law_neo_absolutism)
	requires_law_or = { law_monarchy law_social_monarchy }

	# While this law is active, these can't be enacted. This is what locks you in.
	disallowing_laws = {
		law_presidential_republic
		law_parliamentary_republic
		law_theocracy
		law_council_republic
		law_corporate_state
		law_anarchy
	}

	is_visible = { c:FRA ?= this }  # or has_global_variable = ..., or drop it for all countries

	modifier = {
		country_bureaucrats_pol_str_mult = 0.5
		country_capitalists_pol_str_mult = 0.2
		country_officers_pol_str_mult = 0.2
		country_authority_add = 50
	}

	# copy the technocracy bookkeeping so the switch behaves like the original
	on_activate = {
		if = {
			limit = { has_law = law_type:law_elected_bureaucrats }
			deactivate_law = law_type:law_elected_bureaucrats
			activate_law = law_type:law_appointed_bureaucrats
		}
		scripted_effect_parties_disappearence = yes
		cleanup_voting_strength_modifiers = yes
	}
}
