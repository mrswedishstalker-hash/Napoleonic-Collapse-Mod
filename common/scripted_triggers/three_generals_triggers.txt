# ================= JEAN =================
posture_jean_army = {
	NOT = { any_country = { army_size > root.army_size } }
}
posture_jean_navy = {
	NOT = { any_country = { navy_size > root.navy_size } }
}
posture_jean_defeat_gbr = {
	has_variable = generals_defeated_gbr
}
posture_jean_top_power = {
	country_rank >= rank_value:great_power
	global_country_ranking = 1
}
posture_jean_path_met = {
	posture_jean_army = yes
	posture_jean_navy = yes
	OR = {
		posture_jean_defeat_gbr = yes
		posture_jean_top_power = yes
	}
}
domestic_jean_army = {
	has_law = law_type:law_professional_army
}
domestic_jean_navy = {
	has_law = law_type:law_professional_navy
}
domestic_jean_secret_police = {
	has_law = law_type:law_secret_police 
}
domestic_jean_clout_20 = {
	ig:ig_armed_forces ?= { ig_clout >= 0.2 }
}
domestic_jean_path_met = {
	domestic_jean_army = yes
	domestic_jean_navy = yes
	OR = {
		domestic_jean_secret_police = yes
		domestic_jean_clout_20 = yes
	}
}
economy_jean_barrack = {
	any_scope_state = {
		filter = { is_incorporated = yes }
		has_building = building_barrack
		percent = 1
	}
}
economy_jean_market = {
	NOT = {
	any_scope_state = {
		filter = { is_incorporated = yes }
		state_infrastructure_balance < 0 	
	}
	}
}
economy_jean_army = {
	NOT = {
		any_scope_building = {
			is_building_type = building_barrack
			building_has_goods_shortage = yes
		}
	}
}
economy_jean_shipyard = {
	var:jg_shipyards_built >= 10
}
economy_jean_path_met = {
	economy_jean_army = yes 
	economy_jean_market = yes 
	OR = {
		economy_jean_barrack = yes 
		economy_jean_shipyard = yes
	}
}

# ================= REILLE =================
posture_reille_gdp = {
	gdp_ranking <= 3
}
posture_reille_wide_investment = {
    any_country = {
        count >= 2
        NOT = { this = root }
        root = {
            has_treaty_foreign_investment_rights_with = {
                TARGET = prev
            }
        }
    }
}
posture_reille_trade_privileges = {
	any_country = {
		NOT = { this = root }
		any_scope_treaty = {
			binds = root
			any_scope_article = {
				has_type = trade_privilege
			}
		}
	}
}
posture_reille_wide_investment = {
	any_country = {
		count >= 2
		NOT = { this = root }
		root = {
			has_treaty_foreign_investment_rights_with = { TARGET = prev }
		}
	}
}
posture_reille_path_met = {
	posture_reille_gdp = yes
	posture_reille_investment = yes
	OR = {
		posture_reille_trade_privileges = yes
		posture_reille_wide_investment = yes
	}
}
