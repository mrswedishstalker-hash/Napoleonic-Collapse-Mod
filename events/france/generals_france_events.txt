je_generals.2 = {
    type = country_event
    title = "je_generals.2.t"
    desc = "je_generals.2.d"
    flavor = "je_generals.2.f"
    
    icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"
    
    # Ensures the event only fires for the country running the dynamic narrative
    trigger = {
        has_variable = three_generals_startup
    }

    # =========================================================================
    # OPTION 1: JEAN'S VICTORY (MILITARY PATH)
    # =========================================================================
    option = {
        name = "je_generals.2.a" # "Jean's strategy carries the day!"
        
        # Validates that the conditions for the Military Path were achieved
        trigger = {
            NOT = { any_country = { army_size > root.army_size } }
            NOT = { any_country = { navy_size > root.navy_size } }
            OR = {
                has_country_flag = defeated_great_britain_flag
                AND = {
                    country_rank = rank_value:great_power
                    rank_value:great_power = { is_rank_1 = yes }
                }
            }
        }

        # Progresses Jean's bar. Change 'change_scripted_bar_progress' or value 
        # to match whatever your mod's script engine uses for progress tracking.
        change_scripted_bar_progress = {
            bar = generals_jean_bar
            value = 33 # Adjust this value based on how much of a boost you want to give
        }
        
        # Flavorful roleplay modifiers
        #add_modifier = { name = military_dominance_buff duration = 3650 }
    }

    # =========================================================================
    # OPTION 2: BERTRAND'S VICTORY (GEOPOLITICAL PATH)
    # =========================================================================
    option = {
        name = "je_generals.2.b" # "Bertrand's grand strategy secures our position."
        
        # Validates that the conditions for the Geopolitical Path were achieved
        trigger = {
            has_power_bloc = yes
            power_bloc = { is_power_bloc_type = pb_type_sovereign_empire }
            has_rivalry = c:GBR
            has_rivalry = c:AUT
            has_rivalry = c:TUR
            OR = {
                has_law = law_type:law_mercantilism
                NOT = {
                    any_country = {
                        market_goods_produced_value = {
                            goods = g:grain
                            value > root.market_goods_produced_value = { goods = g:grain }
                        }
                    }
                }
            }
        }

        # Progresses Bertrand's bar
        change_scripted_bar_progress = {
            bar = generals_bertrand_bar
            value = 33
        }
        
        #add_modifier = { name = imperial_prestige_buff duration = 3650 }
    }

    # =========================================================================
    # OPTION 3: REILLE'S VICTORY (ECONOMIC PATH)
    # =========================================================================
    option = {
        name = "je_generals.2.c" # "Reille's economic mastery takes hold."
        
        # Validates that the conditions for the Economic Path were achieved
        trigger = {
            NOT = {
                count_countries = {
                    limit = { gdp > root.gdp }
                    count >= 3
                }
            }
            any_country = {
                NOT = { this = root }
                root = { has_investment_rights_in = prev }
            }
            OR = {
                any_country = {
                    NOT = { this = root }
                    root = { has_trade_privileges_in = prev }
                }
                value:urban_and_resource_gdp_share >= 40
            }
        }

        # Progresses Reille's bar
        change_scripted_bar_progress = {
            bar = generals_reille_bar
            value = 33
        }
        
        #add_modifier = { name = industrial_juggernaut_buff duration = 3650 }
    }
}
